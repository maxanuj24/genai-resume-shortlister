# CVSortingPipeline – Terminal Execution Steps

# 1. Go to Project Folder
cd Capstone_Project-CS01/Codebase

# 2. Create Python Virtual Environment
python3 -m venv .venv

# 3. Activate Virtual Environment
source .venv/bin/activate

# 4. Install Python Dependencies (Manually Installed)
# All dependencies from requirements.txt have been manually installed:
# - pandas==2.2.2
# - numpy==1.26.4
# - pdfminer.six==20231228
# - python-docx==1.1.0
# - requests==2.31.0
# - sentence-transformers==2.7.0
# - torch==2.2.2
#
# If needed, run: pip install -r requirements.txt

# 5. Install Ollama (macOS)
brew install ollama

# 6. Start Ollama Service (runs in background)
ollama serve

# 7. Download Required Models
ollama pull gemma:2b
ollama pull llama3:8b

# 8. Verify Models
ollama list

# 9. Run CV Sorting Pipeline with Default Folders
python main.py

# Optional: Run with Custom Arguments
python main.py \
--resume_folder ./resumes \
--jd_file ./job_description.txt \
--top_k 5 \
--output_file ./results.csv

# 10. Check Output CSV
cat results.csv

# 11. Check Help / Usage
python main.py -h
